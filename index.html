let currentQuestions = []; let currentMode = 'study'; // دو حالت: study (مطالعه ۵ نکته) یا quiz (حل تست‌ها) let currentQIndex = 0; let userScore = 0; let userRole = 'ثبت نشده'; let activeNodeElement = null; let currentChapterData = null; let userAnswers = []; // رندر کردن اتوماتیک نقشه راه function renderRoadmap() { const container = document.getElementById('roadmap-container'); let currentWorld = 0; let pathContainer = null; let isFirstNode = true; millerRoadmap.forEach((ch) => { if (ch.w !== currentWorld) { currentWorld = ch.w; const section = document.createElement('div'); section.className = 'level-section'; section.innerHTML = `
${worldNames[currentWorld]}
`; pathContainer = document.createElement('div'); pathContainer.className = 'path-container'; section.appendChild(pathContainer); container.appendChild(section); } const node = document.createElement('div'); node.className = `level-node ${isFirstNode ? '' : 'locked'}`; if (ch.type === "dry") { node.classList.add('dry-node'); node.style.color = "white"; } else { node.style.backgroundColor = `var(--w${currentWorld}-color)`; node.style.color = "white"; } node.innerHTML = ` ${ch.icon}
فصل ${ch.id}: ${ch.title}
`; node.onclick = function() { if(!this.classList.contains('locked')) startQuiz(ch, this); }; pathContainer.appendChild(node); isFirstNode = false; }); } function showScreen(screenId) { document.querySelectorAll('.screen').forEach(el => el.classList.remove('active')); document.getElementById(screenId).classList.add('active'); } // خواندن شناسه‌های موعد امروز از آدرس URL const urlParams = new URLSearchParams(window.location.search); const rawReviewIds = urlParams.get('review_ids'); const dueReviewQuestionIds = rawReviewIds ? rawReviewIds.split(',').filter(x => x.length > 0).map(Number) : []; function checkDailyReviewBanner() { const banner = document.getElementById('review-banner'); if (banner && dueReviewQuestionIds.length > 0) { banner.style.display = 'block'; const countText = document.getElementById('review-count-text'); if (countText) countText.innerText = `امروز ${dueReviewQuestionIds.length} نکته در نوبت مرور حافظه بلندمدت داری.`; } } function finishOnboarding(role) { userRole = role; renderRoadmap(); showScreen('dashboard-screen'); document.getElementById('bottom-nav').style.display = 'flex'; checkDailyReviewBanner(); } function switchTab(tabId, btn) { showScreen(tabId); document.querySelectorAll('.nav-item').forEach(el => el.classList.remove('active')); btn.classList.add('active'); } // لود مستقیم فایل جیسون مربوط به هر فصل async function startQuiz(chapterData, element) { activeNodeElement = element; currentChapterData = chapterData; document.getElementById('bottom-nav').style.display = 'none'; currentMode = 'study'; // تنظیم شروع روی حالت مطالعه نکات currentQIndex = 0; userScore = 0; userAnswers = []; document.getElementById('question-text').innerText = "در حال بارگذاری سوالات... ⏳"; document.getElementById('options-container').innerHTML = ''; showScreen('quiz-screen'); try { const fileName = chapterData.id === 16 ? 'airway_managment.json' : `chapter_${chapterData.id}.json`; const cacheBuster = new Date().getTime(); const response = await fetch(`${fileName}?t=${cacheBuster}`, { headers: { 'Cache-Control': 'no-cache' } }); if (!response.ok) { throw new Error(`خطای دریافت فایل (${response.status}): مطمئن شوید فایل "${fileName}" در کنار index.html آپلود شده است.`); } currentQuestions = await response.json(); currentQuestions = currentQuestions.sort(() => 0.5 - Math.random()).slice(0, 5); if (!Array.isArray(currentQuestions) || currentQuestions.length === 0) { throw new Error("فایل جیسون خالی است یا ساختار آرایه ندارد."); } loadQuestion(); } catch (error) { console.error("Quiz Load Error:", error); document.getElementById('question-text').innerText = `⚠️ خطا در بارگذاری فصل ${chapterData.id}:\n${error.message}`; document.getElementById('options-container').innerHTML = ` `; } } // کنترل ورود به فاز مطالعه یا فاز آزمون function loadQuestion() { if (currentMode === 'study') { loadStudyTip(); } else { loadQuizQuestion(); } } // فاز ۱: نمایش ۵ نکته به صورت متوالی function loadStudyTip() { const total = currentQuestions.length; const progressPercent = ((currentQIndex + 1) / total) * 100; document.getElementById('progress').style.width = progressPercent + '%'; const q = currentQuestions[currentQIndex]; document.getElementById('question-text').innerHTML = `
💡 نکته آموزشی (${currentQIndex + 1} از ${total}):
${q.tip_text}
`; const isLastTip = (currentQIndex === total - 1); const btnText = isLastTip ? "شروع آزمون این ۵ نکته 🎯" : "نکته بعدی ⬅️"; const btnAction = isLastTip ? "startQuizPhase()" : "nextTip()"; document.getElementById('options-container').innerHTML = ` `; } function nextTip() { currentQIndex++; loadStudyTip(); } function startQuizPhase() { currentMode = 'quiz'; currentQIndex = 0; document.getElementById('progress').style.width = '0%'; loadQuizQuestion(); } // فاز ۲: آزمون ۵ تستی function loadQuizQuestion() { if (currentQIndex >= currentQuestions.length) { return finishQuiz(); } const progressPercent = ((currentQIndex) / currentQuestions.length) * 100; document.getElementById('progress').style.width = progressPercent + '%'; const q = currentQuestions[currentQIndex]; document.getElementById('question-text').innerHTML = `
تست ${currentQIndex + 1} از ${currentQuestions.length}
❓ ${q.question}
`; const optsContainer = document.getElementById('options-container'); optsContainer.innerHTML = ''; q.options.forEach((opt, idx) => { const btn = document.createElement('button'); btn.className = 'btn'; btn.innerText = opt; btn.onclick = () => checkAnswer(idx, btn); optsContainer.appendChild(btn); }); } function checkAnswer(selectedIndex, btnNode) { const q = currentQuestions[currentQIndex]; const allBtns = document.getElementById('options-container').children; for (let b of allBtns) b.disabled = true; const correctIdx = q.correct_option_index !== undefined ? q.correct_option_index : q.correct_index; const isCorrect = (selectedIndex === correctIdx); const questionId = q.id || (currentChapterData.id * 1000 + currentQIndex + 1); userAnswers.push({ question_id: questionId, is_correct: isCorrect }); let delayTime = 1200; if (selectedIndex === correctIdx) { btnNode.classList.add('correct'); userScore += 10; document.getElementById('board-score').innerText = userScore + " XP"; try { tg.HapticFeedback.notificationOccurred('success'); } catch(e){} } else { btnNode.classList.add('wrong'); if (allBtns[correctIdx]) allBtns[correctIdx].classList.add('correct'); const expDiv = document.createElement('div'); expDiv.style.marginTop = "20px"; expDiv.style.color = "#e53935"; expDiv.style.fontSize = "16px"; expDiv.style.fontWeight = "bold"; expDiv.style.lineHeight = "1.6"; expDiv.innerText = "❌ " + q.explanation; document.getElementById('options-container').appendChild(expDiv); delayTime = 4000; try { tg.HapticFeedback.notificationOccurred('error'); } catch(e){} } setTimeout(() => { currentQIndex++; loadQuizQuestion(); }, delayTime); }
